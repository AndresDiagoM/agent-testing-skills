# Template: `auth.setup.ts`

Code template shipped as Markdown in the directory edition (full edition: `assets/kit/e2e/auth.setup.ts`).
Copy the code below into the user's repository as `auth.setup.ts` and adapt it as the references say.

```ts
/**
 * auth.setup.ts: the ONLY UI logins of the suite besides the auth spec. Run by the `setup` project (see
 * playwright.config.snippet.ts); every other project starts from the saved storageState. Never use globalSetup to log in.
 *
 * Roles (one setup test each, run in parallel). Source, first hit wins:
 *   1. kit.config.json  e2e.authRoles: [{ name, loginUrl, statePath?, emailLabel?, passwordLabel?, submitName?, successUrl? }]
 *      (found by walking up from the cwd; E2E_KIT_CONFIG points at another file). The label fields are regex sources,
 *      case-insensitive; successUrl is a regex source for the URL after login.
 *   2. a single default role "owner" at /login.
 *
 * Credentials: ONLY from the environment, never literals in this file or in kit.config.json:
 *   E2E_<ROLE>_EMAIL, E2E_<ROLE>_PASSWORD   (ROLE = role name upper-cased, non-alphanumerics become "_")
 *
 * Per-spec users (optional): kit.config.json e2e.specUsers: ["checkout", "reports"]. One throwaway user per spec file
 * that writes data, created fresh by `provisionUser` below, signed in once, state saved per spec. Needs:
 *   E2E_RUN_ID           set by the config snippet (one id per run), so two runs on one backend never collide.
 *   E2E_SPEC_PASSWORD    generated by the config snippet at run time (random), shared with the workers through the
 *                        environment. Never write one here.
 *   E2E_EMAIL_DOMAIN     address domain, default "example.test".
 * The user is addressed e2e-<spec>-<runId>@<domain>. Delete these users in a teardown project (see README.md).
 *
 * Other: E2E_AUTH_DIR (state folder, default e2e/.auth, git-ignore it), E2E_READY_TIMEOUT_MS (via goto-ready.ts).
 * State files are named <name>-<runId>.json (or the role's statePath with -<runId> before the extension).
 */
import { existsSync, readFileSync } from 'node:fs';
import path from 'node:path';
import { test as setup, expect, type Page } from '@playwright/test';
import { gotoReady } from './goto-ready';

interface AuthRole {
  name: string;
  loginUrl?: string;
  statePath?: string;
  emailLabel?: string;
  passwordLabel?: string;
  submitName?: string;
  successUrl?: string;
}
interface KitE2e {
  authRoles?: AuthRole[];
  specUsers?: string[];
}

const LOGIN_TIMEOUT = 45_000; // post-login navigation; generous on purpose (Windows margin), costs time only on failure

function readKitE2e(): KitE2e {
  let dir = process.cwd();
  const explicit = process.env.E2E_KIT_CONFIG;
  for (;;) {
    const file = explicit ?? path.join(dir, 'kit.config.json');
    if (existsSync(file)) {
      const parsed = JSON.parse(readFileSync(file, 'utf8')) as { e2e?: KitE2e };
      return parsed.e2e ?? {};
    }
    const parent = path.dirname(dir);
    if (explicit || parent === dir) return {};
    dir = parent;
  }
}

const kit = readKitE2e();
const roles: AuthRole[] = kit.authRoles?.length ? kit.authRoles : [{ name: 'owner' }];
const specUsers: string[] = kit.specUsers ?? [];

function requireEnv(name: string, why: string): string {
  const value = process.env[name];
  if (!value) throw new Error(`${name} is not set (${why}). Export it before running; it is never stored in the repo.`);
  return value;
}

function runId(): string {
  const id = requireEnv('E2E_RUN_ID', 'the config snippet sets it once per run').trim().toLowerCase();
  if (!/^[a-z0-9]{3,16}$/.test(id)) throw new Error(`E2E_RUN_ID "${id}" must be 3-16 lowercase letters or digits.`);
  return id;
}

/** Same formula as playwright.config.snippet.ts: the project's `storageState` must point at this file. */
function statePathFor(name: string, explicit?: string): string {
  const run = runId();
  const file = explicit
    ? explicit.replace(/(\.json)?$/, `-${run}.json`)
    : path.join(process.env.E2E_AUTH_DIR || 'e2e/.auth', `${name}-${run}.json`);
  return path.resolve(process.cwd(), file);
}

const envKey = (name: string): string => name.toUpperCase().replace(/[^A-Z0-9]/g, '_');

async function signInAndSave(page: Page, role: AuthRole, email: string, password: string, statePath: string): Promise<void> {
  const loginUrl = role.loginUrl ?? '/login';
  // gotoReady waits for hydration: a click before it submits the form as a plain request and stays on the login page.
  await gotoReady(page, loginUrl);
  await page.getByLabel(new RegExp(role.emailLabel ?? 'e-?mail', 'i')).fill(email);
  await page.getByLabel(new RegExp(role.passwordLabel ?? 'password', 'i')).fill(password);
  await page.getByRole('button', { name: new RegExp(role.submitName ?? 'sign in|log in|login', 'i') }).click();
  if (role.successUrl) {
    await expect(page).toHaveURL(new RegExp(role.successUrl), { timeout: LOGIN_TIMEOUT });
  } else {
    const loginPath = loginUrl.split('?')[0] ?? loginUrl;
    await expect(page).not.toHaveURL((url) => url.pathname === loginPath, { timeout: LOGIN_TIMEOUT });
  }
  await page.context().storageState({ path: statePath });
}

for (const role of roles) {
  setup(`sign in as ${role.name} and save the session`, async ({ page }) => {
    const key = envKey(role.name);
    const email = requireEnv(`E2E_${key}_EMAIL`, `address of the ${role.name} account`);
    const password = requireEnv(`E2E_${key}_PASSWORD`, `password of the ${role.name} account`);
    await signInAndSave(page, role, email, password, statePathFor(role.name, role.statePath));
  });
}

/**
 * PROJECT-SPECIFIC HOOK. Create the user from scratch (so a leftover of an aborted run cannot leak in): admin API call,
 * seed script or SDK of your backend. Use the generated password, never a literal. Only called for `e2e.specUsers`.
 */
async function provisionUser(_email: string, _password: string): Promise<void> {
  throw new Error('provisionUser is a template hook: implement it for this backend (admin API, SDK or seed script).');
}

for (const spec of specUsers) {
  setup(`provision and sign in the e2e user of ${spec}`, async ({ page }) => {
    const email = `e2e-${spec}-${runId()}@${process.env.E2E_EMAIL_DOMAIN || 'example.test'}`;
    const password = requireEnv('E2E_SPEC_PASSWORD', 'the config snippet generates it per run');
    await provisionUser(email, password);
    await signInAndSave(page, roles[0] ?? { name: 'owner' }, email, password, statePathFor(spec));
  });
}
```
